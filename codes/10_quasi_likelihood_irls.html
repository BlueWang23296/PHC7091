IRLS estimates $\beta$ using a working response and weights. For $V(\mu)=\mu(1-\mu)$ and logit link, $g'(\mu)=1/[\mu(1-\mu)]$ and $W_i=1/[V(\mu_i)g'(\mu_i)^2]$.

v.mu <- function(mu) mu * (1 - mu)
g.first <- function(mu) 1 / (mu * (1 - mu))

mu.hat <- (y + mean(y)) / 2
W0 <- diag(as.vector(1 / (v.mu(mu.hat) * g.first(mu.hat)^2)))
z0 <- log(mu.hat / (1 - mu.hat)) + g.first(mu.hat) * (y - mu.hat)

beta1 <- solve(t(X) %*% W0 %*% X) %*% t(X) %*% W0 %*% z0
beta0 <- beta1 + 1

while (max(abs(beta0 - beta1)) > 0.000001) {
  beta0 <- beta1
  eta0 <- X %*% beta0
  mu.hat <- exp(eta0) / (1 + exp(eta0))
  z0 <- eta0 + g.first(mu.hat) * (y - mu.hat)
  W0 <- diag(as.vector(1 / (v.mu(mu.hat) * g.first(mu.hat)^2)))
  beta1 <- solve(t(X) %*% W0 %*% X) %*% t(X) %*% W0 %*% z0
}

print(beta1)

After fitting, estimate scale and Pearson residuals.

y.fit <- exp(X %*% beta1) / (1 + exp(X %*% beta1))
sigma <- sum((y - y.fit)^2 / v.mu(y.fit)) / 72

var.beta <- sigma * solve(t(X) %*% W0 %*% X)
sd.beta <- sqrt(diag(var.beta))

r.p <- (y - y.fit) / sqrt(v.mu(y.fit) * sigma)
plot(log(y.fit / (1 - y.fit)), r.p,
     xlab = "eta", ylab = "Pearson's residual")
abline(h = 0, lty = 2)
